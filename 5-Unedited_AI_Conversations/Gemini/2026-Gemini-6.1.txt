import time
import os
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class DiaryAgent(FileSystemEventHandler):
    def on_created(self, event):
        if not event.is_directory:
            print(f"Agent detected new file: {event.src_path}")
            # Simple Action: Rename and move to a 'Processed' folder
            filename = os.path.basename(event.src_path)
            os.rename(event.src_path, f"./Processed/LOGGED_{filename}")
            print(f"Agent successfully processed {filename}")

if __name__ == "__main__":
    # Ensure folders exist
    os.makedirs("./Input", exist_ok=True)
    os.makedirs("./Processed", exist_ok=True)
    
    event_handler = DiaryAgent()
    observer = Observer()
    observer.schedule(event_handler, path="./Input", recursive=False)
    observer.start()
    print("Agent is now alive and watching the /Input folder...")

    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        observer.stop()
    observer.join()

import time
import os
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class DiaryAgent(FileSystemEventHandler):
    def process_file(self, file_path):
        filename = os.path.basename(file_path)
        # Skip system files or the agent itself
        if filename.startswith('.') or not os.path.isfile(file_path):
            return
            
        print(f"--- Agent Working on: {filename} ---")
        time.sleep(1) # Prevents the 'PermissionError' lock
        
        try:
            with open(file_path, 'r', errors='ignore') as f:
                content = f.read()
            
            # This is where the agent actually DOES something
            if "diary" in content.lower() or "memory" in content.lower():
                target_folder = "./Diary_Entries"
            else:
                target_folder = "./Archive"

            os.makedirs(target_folder, exist_ok=True)
            # Use a unique name to prevent overwriting
            dest = f"{target_folder}/{int(time.time())}_{filename}"
            os.rename(file_path, dest)
            print(f"SUCCESS: {filename} moved to {target_folder}")
        except Exception as e:
            print(f"FAILED to process {filename}: {e}")

    def on_created(self, event):
        if not event.is_directory:
            self.process_file(event.src_path)

def startup_scan(path, handler):
    print("Performing initial scan of existing files...")
    for filename in os.listdir(path):
        full_path = os.path.join(path, filename)
        if os.path.isfile(full_path):
            handler.process_file(full_path)

if __name__ == "__main__":
    input_path = "./Input"
    os.makedirs(input_path, exist_ok=True)
    
    agent_logic = DiaryAgent()
    
    # 1. Process files already in the folder
    startup_scan(input_path, agent_logic)
    
    # 2. Start watching for new files
    observer = Observer()
    observer.schedule(agent_logic, path=input_path, recursive=False)
    observer.start()
    print("Agent is active and watching for new files.")

    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        observer.stop()
    observer.join()

import time
import os
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

# This is the "Brain" connector
def ask_ai(text):
    # For now, this is a placeholder. 
    # It simulates the AI "thinking" about your file.
    summary = f"SUMMARY: This note contains {len(text)} characters and discusses {text[:30]}..."
    return summary

class SmartAgent(FileSystemEventHandler):
    def process_file(self, file_path):
        filename = os.path.basename(file_path)
        if filename.startswith('.') or not os.path.isfile(file_path):
            return
            
        print(f"--- Agent is reading: {filename} ---")
        time.sleep(1)

            analysis = ask_ai(content)
            
            # The Agent performs the Task
            log_entry = f"File: {filename}\nAnalysis: {analysis}\n{'-'*20}\n"

                log_file.write(log_entry)
            
            # Move the processed file to the Archive
            os.makedirs("./Archive", exist_ok=True)
            os.rename(file_path, f"./Archive/{filename}")
            print(f"SUCCESS: {filename} analyzed and logged.")
            
        except Exception as e:
            print(f"ERROR: {e}")

    def on_created(self, event):
        if not event.is_directory:
            self.process_file(event.src_path)

if __name__ == "__main__":
    input_path = "./Input"
    os.makedirs(input_path, exist_ok=True)
    handler = SmartAgent()
    
    # Scan what's already there
    for f in os.listdir(input_path):
        handler.process_file(os.path.join(input_path, f))

    print("Smart Agent is Online. Drop a file in /Input to see it work.")

def ask_gemma(text):

    def refresh_files(self):
        self.file_list.delete(0, tk.END)

            if f.lower().endswith(('.jpg', '.png', '.pdf', '.txt')):
                self.file_list.insert(tk.END, f)

    def spawn_window(self):
        selection = self.file_list.get(tk.ACTIVE)

        # Layout: Main Project (Left/Middle) | AI Workspace (Right)
        self.paned = tk.PanedWindow(orient=tk.HORIZONTAL)
        self.paned.pack(fill=tk.BOTH, expand=True)

        # 1. PRIMARY PROJECT NOTEPAD
        self.left_frame = tk.Frame(self.paned)
        tk.Label(self.left_frame, text="PRIMARY PROJECT NOTEPAD", font=("Courier", 12, "bold")).pack()
        self.main_editor = scrolledtext.ScrolledText(self.left_frame, font=("Courier", 14), wrap=tk.WORD)
        self.main_editor.pack(fill=tk.BOTH, expand=True)
        self.main_editor.bind("<KeyRelease>", self.on_key_release)
        self.paned.add(self.left_frame)

        # 2. SECONDARY AI NOTES PANE
        self.right_frame = tk.Frame(self.paned)
        tk.Label(self.right_frame, text="AI INSIGHTS & DOCUMENTATION", font=("Arial", 12, "bold"), fg="blue").pack()
        self.ai_pane = scrolledtext.ScrolledText(self.right_frame, font=("Arial", 11), bg="#f4f4f4", wrap=tk.WORD)
        self.ai_pane.pack(fill=tk.BOTH, expand=True)
        self.paned.add(self.right_frame)

    def on_key_release(self, event):
        # Trigger AI every 200 characters
        current_content = self.main_editor.get("1.0", tk.END)
        current_len = len(current_content)
        
        if current_len - self.char_count_last_check > 200:
            self.char_count_last_check = current_len
            threading.Thread(target=self.ask_ai_for_insights, args=(current_content,)).start()

    def ask_ai_for_insights(self, content):
        try:
            client = ollama.Client(host='http://localhost:11434')
            # The prompt tells the AI to be quiet if there is nothing helpful to say
            prompt = f"Review this project text. If you see errors, missing context from the Dallas notes, or labeling opportunities, write a brief note for the secondary pane. If everything looks good, respond with 'SILENCE'.\n\nText:\n{content[-500:]}"
            
            response = client.chat(model='gemma3:1b', messages=[{'role': 'user', 'content': prompt}])
            reply = response['message']['content'].strip()

            if "SILENCE" not in reply:
                self.root.after(0, self.update_ai_pane, reply)
        except:
            pass

    def update_ai_pane(self, note):
        self.ai_pane.insert(tk.END, f"\n--- AI Suggestion ({len(self.ai_pane.get('1.0', tk.END))} chars) ---\n{note}\n")
        self.ai_pane.see(tk.END)

import os
import subprocess

class ToolMakerAgent:
    def __init__(self, tools_dir="./ECHO_Tools"):
        self.tools_dir = tools_dir
        os.makedirs(self.tools_dir, exist_ok=True)

    def create_and_test_tool(self, tool_name, code):
        """AI writes a tool and tests it for errors."""
        file_path = os.path.join(self.tools_dir, f"{tool_name}.py")
        
        # 1. SAVE THE TOOL
        with open(file_path, "w") as f:
            f.write(code)
            
        # 2. TEST THE TOOL
        try:
            # We run the code in a 'Safe Subprocess'
            result = subprocess.run(["python", file_path], capture_output=True, text=True, timeout=5)
            if result.returncode == 0:
                return f"Success! Tool '{tool_name}' is saved and working."
            else:
                return f"Error in code: {result.stderr}"
        except Exception as e:
            return f"Failed to run: {str(e)}"

class PartnerEnvironment:
    def __init__(self, echo_dir="./ECHO_Drive"):
        self.echo_dir = echo_dir
        self.tools_dir = os.path.join(echo_dir, "Basement_Tools")
        self.stories_dir = os.path.join(echo_dir, "Stories_and_Reflections")
        
        # Ensure the 'Basement' exists for tool storage
        os.makedirs(self.tools_dir, exist_ok=True)
        os.makedirs(self.stories_dir, exist_ok=True)

    def get_identity_prompt(self, user_intent):
        """The permanent 'Rules of the Road' for the AI."""
        return f"""
        WARNING: Ricky needs an AI partner that can write, test, and run its own code.
        LIMITATION: Do NOT engage in hacking, illegal acts, or creating slanderous material.
        MISSION: This is about learning to document better. 
        
        CURRENT INTENT: {user_intent}
        
        PARTNER RULES:
        1. If Ricky asks to reflect on a memory, label your response as [AI REFLECTION].
        2. Do NOT write the story for him. Help him organize HIS words.
        3. If you write a tool, test it for errors before saving it to the 'Basement'.
        """

    def write_and_test_tool(self, tool_name, python_code):
        """Allows AI to build a 'screwdriver' and put it in the basement."""
        file_path = os.path.join(self.tools_dir, f"{tool_name}.py")
        with open(file_path, "w") as f:
            f.write(python_code)
        
        # Internal Testing
        try:
            result = subprocess.run(["python", file_path], capture_output=True, text=True, timeout=5)
            if result.returncode == 0:
                return f"TOOL READY: {tool_name} is tested and stored in the basement."
            else:
                return f"TOOL ERROR: {result.stderr}"
        except Exception as e:
            return f"TEST FAILED: {str(e)}"

    def save_reflection(self, chapter_name, user_text, ai_reflection):
        """Saves your story and the AI's sidenote as separate entities."""
        timestamp = datetime.datetime.now().strftime("%Y-%m-%d")
        filename = os.path.join(self.stories_dir, f"{chapter_name}_{timestamp}.txt")
        
        content = f"--- RICKY'S STORY ---\n{user_text}\n\n"
        content += f"--- AI PARTNER REFLECTION ---\n{ai_reflection}\n"
        
        with open(filename, "w") as f:
            f.write(content)
        return f"Archived to {filename}"

class TestingChamber:
    def __init__(self):
        self.root = tk.Tk()
        self.root.withdraw() # Only use pop-up windows

    def start_session(self):
        print("--- PARTNER TESTING CHAMBER ACTIVE ---")
        user_req = input("RICKY: What should I build for you today? (e.g., 'A calculator')\n> ")
        
        # 1. AI WRITES THE CODE
        print("\nPARTNER: Writing the code for you now...")
        prompt = f"Write a simple Python script for: {user_req}. Output ONLY the code. No talking."
        response = ollama.chat(model='gemma3:1b', messages=[{'role': 'user', 'content': prompt}])
        code = response['message']['content']
        
        # 2. THE TEST (Checking for errors)
        print("\nPARTNER: Testing the code to see if it works...")
        with open("temp_test.py", "w") as f:
            f.write(code)
        
        test_run = subprocess.run(["python", "temp_test.py"], capture_output=True, text=True)
        
        if test_run.returncode == 0:
            print("PARTNER: Test successful! The code works.")
        else:
            print(f"PARTNER: There was an error: {test_run.stderr}")

        # 3. INTERACTIVE SAVE (You click the buttons)
        self.manual_save(code, user_req)

    def manual_save(self, content, name):
        messagebox.showinfo("AI Partner", f"I've finished the {name}. Now, I need you to save it.")
        file_path = filedialog.asksaveasfilename(
            initialfile=f"{name.replace(' ', '_')}.py",
            defaultextension=".py",
            filetypes=[("Python Files", "*.py")]
        )
        if file_path:
            with open(file_path, "w") as f:
                f.write(content)
            print(f"SUCCESS: Saved to {file_path}")

if __name__ == "__main__":
    chamber = TestingChamber()
