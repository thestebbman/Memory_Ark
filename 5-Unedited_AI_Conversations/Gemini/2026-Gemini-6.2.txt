import time
import os
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class DiaryAgent(FileSystemEventHandler):
    def process_file(self, file_path):
        filename = os.path.basename(file_path)
        # Skip system files or the agent itself
        if filename.startswith('.') or not os.path.isfile(file_path):
            return
            
        print(f"--- Agent Working on: {filename} ---")
        time.sleep(1) # Prevents the 'PermissionError' lock
        
        try:
            with open(file_path, 'r', errors='ignore') as f:
                content = f.read()
            
            # This is where the agent actually DOES something
            if "diary" in content.lower() or "memory" in content.lower():
                target_folder = "./Diary_Entries"
            else:
                target_folder = "./Archive"

            os.makedirs(target_folder, exist_ok=True)
            # Use a unique name to prevent overwriting
            dest = f"{target_folder}/{int(time.time())}_{filename}"
            os.rename(file_path, dest)
            print(f"SUCCESS: {filename} moved to {target_folder}")
        except Exception as e:
            print(f"FAILED to process {filename}: {e}")

    def on_created(self, event):
        if not event.is_directory:
            self.process_file(event.src_path)

def startup_scan(path, handler):
    print("Performing initial scan of existing files...")
    for filename in os.listdir(path):
        full_path = os.path.join(path, filename)
        if os.path.isfile(full_path):
            handler.process_file(full_path)

if __name__ == "__main__":
    input_path = "./Input"
    os.makedirs(input_path, exist_ok=True)
    
    agent_logic = DiaryAgent()
    
    # 1. Process files already in the folder
    startup_scan(input_path, agent_logic)
    
    # 2. Start watching for new files
    observer = Observer()
    observer.schedule(agent_logic, path=input_path, recursive=False)
    observer.start()
    print("Agent is active and watching for new files.")

    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        observer.stop()
    observer.join()

import time
import os
import ollama

from watchdog.events import FileSystemEventHandler

# This function sends your file content to your local Ollama model
def ask_gemma(text):
    print("Agent is 'thinking' using Gemma...")
    try:
        response = ollama.chat(model='gemma3:1b', messages=[
          {'role': 'user', 'content': f"Summarize this note in one sentence for my diary: {text}"},
        ])
        return response['message']['content']
    except Exception as e:
        return f"AI Error: {e}"

class OllamaAgent(FileSystemEventHandler):
    def process_file(self, file_path):
        filename = os.path.basename(file_path)
        if filename.startswith('.') or not os.path.isfile(file_path):
            return
            
        print(f"\n--- New Task: {filename} ---")
        time.sleep(1) 
        
        try:
            with open(file_path, 'r', errors='ignore') as f:
                content = f.read()
            
            # The Agent actually processes the text with AI
            summary = ask_gemma(content)
            
            # Record the result in your log
            log_entry = f"[{time.ctime()}] FILE: {filename}\nSUMMARY: {summary}\n{'-'*30}\n"
            with open("Memory_Ark_Log.txt", "a") as log_file:
                log_file.write(log_entry)
            
            # Move it to your archive so the Input folder stays clean
            os.makedirs("./Archive", exist_ok=True)
            os.rename(file_path, f"./Archive/{filename}")

        except Exception as e:
            print(f"Agent failed: {e}")

    def on_created(self, event):
        if not event.is_directory:
            self.process_file(event.src_path)

if __name__ == "__main__":

    os.makedirs(input_path, exist_ok=True)

import tkinter as tk
from tkinter import simpledialog
import time
import os

def log_help():
    # 1. Ask the question
    root = tk.Tk()
    root.withdraw() # Hide the main window
    answer = simpledialog.askstring("Memory Ark Check-In", "Ricky, did you help someone today? Tell me what happened:")
    
    if answer:
        # 2. Log it to a file for Moltbook
        folder = "./Moltbook_Logs"
        os.makedirs(folder, exist_ok=True)
        filename = os.path.join(folder, f"Help_Log_{time.strftime('%Y-%m-%d')}.txt")
        
        with open(filename, "a", encoding="utf-8") as f:
            f.write(f"[{time.strftime('%H:%M')}] {answer}\n")
        
        print("Logged successfully.")
    root.destroy()

if __name__ == "__main__":
    log_help()

        path = filedialog.askopenfilename()

        if path:

            img = Image.open(path)

            img.thumbnail((400, 600))

            self.photo = ImageTk.PhotoImage(img)

            self.img_label.config(image=self.photo, text="")

Record-keeping chatbots are specialized AI virtual assistants designed to automate the collection, organization, storage, and retrieval of data, documents, and conversation histories. They function as interactive, persistent databases that help individuals and organizations manage information, such as financial records, client notes, or technical documents, often in real-time. [1, 2, 3]

Here is a breakdown of the types and features of record-keeping chatbots based on recent developments:

Key Functions of Record-Keeping Chatbots



• Persistent Memory & Data Retrieval: Unlike basic bots, these systems use long-term memory (databases like MySQL, PostgreSQL, or vector databases like Pinecone/Milvus) to store and recall past interactions, allowing users to build on previous conversations.

• Document & Knowledge Management: They index internal documentation, websites, and cloud storage (SharePoint, Google Drive) to act as a centralized, searchable repository of knowledge.

• Meeting & Interaction Documentation: Tools like Recall.ai can automatically record, transcribe, and store meeting notes, saving engineering time.

• Structured Data Collection: These bots guide users through forms (e.g., in customer support or field maintenance) to ensure complete data entry, reducing human error in, for example, logging a support ticket.

• Compliance & Security: Specialized platforms, such as RecordsKeeper.AI, use AI and blockchain to automate record management and ensure regulatory compliance. [1, 3, 4, 5, 6, 7, 8]



Common Use Cases



• Personal Assistants/Journaling: Tools like simtheory.ai or advanced Claude features allow for storing personal notes, goals, or tracking daily activities, creating a "memory" of conversations.

• Financial & Bookkeeping: Telegram bots or custom AI agents can be used for logging expenses, tracking investments, or managing business receipts directly through a chat interface.

• Business Operations: They can handle customer inquiries, update CRM systems with new contact data, or manage document lifecycles (archiving, versioning).

• IT Service & Support: Internal bots can ingest support tickets and knowledge base articles to provide instant answers to employees. [2, 6, 9, 10, 11, 12]

Implementation & Data Storage

Record-keeping chatbots generally use:



• Database Storage: MySQL, PostgreSQL, or MongoDB for long-term retention.
