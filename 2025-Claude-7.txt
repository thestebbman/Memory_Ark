import tkinter as tk
from tkinter import scrolledtext, filedialog, messagebox
import requests
import json
import os
from datetime import datetime

class MemoryArkAssistant:
    def __init__(self, root):
        self.root = root
        self.root.title("Memory Ark Assistant")
        self.root.geometry("1200x800")
        
        # Paths
        self.blog_files_path = r"C:\Users\thest\Desktop\Doc- Ai Partner\blog_files"
        self.output_folder = r"C:\Users\thest\Desktop\Doc- Ai Partner\organized_output"
        self.conversation_folder = r"C:\Users\thest\Desktop\Doc- Ai Partner\conversations"
        
        # Create folders if they don't exist
        os.makedirs(self.output_folder, exist_ok=True)
        os.makedirs(self.conversation_folder, exist_ok=True)
        
        # LM Studio API endpoint
        self.api_url = "http://localhost:1234/v1/chat/completions"
        
        # Conversation history
        self.conversation_history = []
        
        # Setup UI
        self.setup_ui()
        
    def setup_ui(self):
        # Main container
        main_frame = tk.Frame(self.root)
        main_frame.pack(fill=tk.BOTH, expand=True, padx=10, pady=10)
        
        # Left side - Document editor
        left_frame = tk.Frame(main_frame)
        left_frame.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=(0, 5))
        
        tk.Label(left_frame, text="Your Document", font=("Arial", 12, "bold")).pack()
        
        self.document_text = scrolledtext.ScrolledText(
            left_frame, 
            wrap=tk.WORD,
            font=("Arial", 11),
            height=35
        )
        self.document_text.pack(fill=tk.BOTH, expand=True, pady=(5, 10))
        
        # Document buttons
        doc_button_frame = tk.Frame(left_frame)
        doc_button_frame.pack(fill=tk.X)
        
        tk.Button(
            doc_button_frame,
            text="Save Document",
            command=self.save_document,
            bg="#4CAF50",
            fg="white",
            font=("Arial", 10, "bold")
        ).pack(side=tk.LEFT, padx=2)
        
        tk.Button(
            doc_button_frame,
            text="Load File for Context",
            command=self.load_context_file,
            bg="#2196F3",
            fg="white",
            font=("Arial", 10, "bold")
        ).pack(side=tk.LEFT, padx=2)
        
        # Right side - AI conversation
        right_frame = tk.Frame(main_frame)
        right_frame.pack(side=tk.RIGHT, fill=tk.BOTH, expand=True, padx=(5, 0))
        
        tk.Label(right_frame, text="AI Partner", font=("Arial", 12, "bold")).pack()
        
        self.ai_conversation = scrolledtext.ScrolledText(
            right_frame,
            wrap=tk.WORD,
            font=("Arial", 11),
            height=30,
            state=tk.DISABLED
        )
        self.ai_conversation.pack(fill=tk.BOTH, expand=True, pady=(5, 10))
        
        # User input
        input_frame = tk.Frame(right_frame)
        input_frame.pack(fill=tk.X, pady=(0, 10))
        
        self.user_input = tk.Entry(input_frame, font=("Arial", 11))
        self.user_input.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=(0, 5))
        self.user_input.bind("<Return>", lambda e: self.send_message())
        
        tk.Button(
            input_frame,
            text="Send",
            command=self.send_message,
            bg="#4CAF50",
            fg="white",
            font=("Arial", 10, "bold")
        ).pack(side=tk.RIGHT)
        
        # AI buttons
        ai_button_frame = tk.Frame(right_frame)
        ai_button_frame.pack(fill=tk.X)
        
        tk.Button(
            ai_button_frame,
            text="Ask AI About Document",
            command=self.ask_about_document,
            bg="#FF9800",
            fg="white",
            font=("Arial", 10, "bold")
        ).pack(side=tk.LEFT, padx=2)
        
        tk.Button(
            ai_button_frame,
            text="Save Conversation",
            command=self.save_conversation,
            bg="#9C27B0",
            fg="white",
            font=("Arial", 10, "bold")
        ).pack(side=tk.LEFT, padx=2)
        
        tk.Button(
            ai_button_frame,
            text="Clear Conversation",
            command=self.clear_conversation,
            bg="#F44336",
            fg="white",
            font=("Arial", 10, "bold")
        ).pack(side=tk.LEFT, padx=2)
        
        # Status bar
        self.status_label = tk.Label(
            self.root,
            text="Ready. Make sure LM Studio is running on localhost:1234",
            bd=1,
            relief=tk.SUNKEN,
            anchor=tk.W
        )
        self.status_label.pack(side=tk.BOTTOM, fill=tk.X)
        
    def add_to_conversation(self, speaker, message):
        self.ai_conversation.config(state=tk.NORMAL)
        
        if speaker == "AI":
            self.ai_conversation.insert(tk.END, f"\n🤖 AI: ", "ai")
            self.ai_conversation.tag_config("ai", foreground="#2196F3", font=("Arial", 11, "bold"))
        else:
            self.ai_conversation.insert(tk.END, f"\n👤 You: ", "user")
            self.ai_conversation.tag_config("user", foreground="#4CAF50", font=("Arial", 11, "bold"))
        
        self.ai_conversation.insert(tk.END, f"{message}\n")
        self.ai_conversation.see(tk.END)
        self.ai_conversation.config(state=tk.DISABLED)
        
        # Add to history
        self.conversation_history.append({
            "speaker": speaker,
            "message": message,
            "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        })
    
    def send_message(self):
        user_message = self.user_input.get().strip()
        if not user_message:
            return
        
        self.user_input.delete(0, tk.END)
        self.add_to_conversation("You", user_message)
        
        # Get AI response
        self.get_ai_response(user_message)
    
    def get_ai_response(self, user_message, include_document=False):
        try:
            self.status_label.config(text="AI is thinking...")
            self.root.update()
            
            # Build prompt
            prompt = user_message
            
            if include_document:
                document_content = self.document_text.get("1.0", tk.END).strip()
                if document_content:
                    prompt = f"Here's the document I'm working on:\n\n{document_content}\n\nMy question: {user_message}"
            
            # Call LM Studio API
            response = requests.post(
                self.api_url,
                json={
                    "messages": [
                        {
                            "role": "system",
                            "content": "You are a helpful AI assistant helping someone organize their Memory Ark - a comprehensive documentation project. Ask ONE question at a time. Be patient. Help them build connections between people, places, and events. Remember what they tell you."
                        },
                        {
                            "role": "user",
                            "content": prompt
                        }
                    ],
                    "temperature": 0.7,
                    "max_tokens": 500
                },
                timeout=30
            )
            
            if response.status_code == 200:
                ai_message = response.json()["choices"][0]["message"]["content"]
                self.add_to_conversation("AI", ai_message)
                self.status_label.config(text="Ready")
            else:
                self.add_to_conversation("AI", f"Error: Couldn't get response (Status {response.status_code})")
                self.status_label.config(text="Error - Check LM Studio")
                
        except requests.exceptions.ConnectionError:
            messagebox.showerror("Connection Error", "Can't connect to LM Studio. Make sure it's running on localhost:1234")
            self.status_label.config(text="Error - LM Studio not running")
        except Exception as e:
            self.add_to_conversation("AI", f"Error: {str(e)}")
            self.status_label.config(text="Error occurred")
    
    def ask_about_document(self):
        document_content = self.document_text.get("1.0", tk.END).strip()
        if not document_content:
            messagebox.showwarning("No Document", "Write something in the document first")
            return
        
        self.add_to_conversation("You", "[Asked AI to review document]")
        self.get_ai_response("Please review this document and ask me ONE question about it to help me improve or clarify it.", include_document=True)
    
    def load_context_file(self):
        filepath = filedialog.askopenfilename(
            initialdir=self.blog_files_path,
            title="Select file for context",
            filetypes=(("Text files", "*.txt"), ("All files", "*.*"))
        )
        
        if filepath:
            try:
                with open(filepath, 'r', encoding='utf-8') as f:
                    content = f.read()
                
                filename = os.path.basename(filepath)
                self.add_to_conversation("You", f"[Loaded {filename} for context]")
                
                # Send to AI
                self.get_ai_response(f"I just loaded a file called {filename}. Here's what's in it:\n\n{content[:2000]}...\n\nAsk me ONE question about what you see here.")
                
            except Exception as e:
                messagebox.showerror("Error", f"Couldn't load file: {str(e)}")
    
    def save_document(self):
        content = self.document_text.get("1.0", tk.END).strip()
        if not content:
            messagebox.showwarning("Empty Document", "Nothing to save")
            return
        
        filename = filedialog.asksaveasfilename(
            initialdir=self.output_folder,
            defaultextension=".txt",
            filetypes=(("Text files", "*.txt"), ("All files", "*.*"))
        )
        
        if filename:
            try:
                with open(filename, 'w', encoding='utf-8') as f:
                    f.write(content)
                messagebox.showinfo("Saved", f"Document saved to {filename}")
                self.status_label.config(text=f"Saved: {os.path.basename(filename)}")
            except Exception as e:
                messagebox.showerror("Error", f"Couldn't save: {str(e)}")
    
    def save_conversation(self):
        if not self.conversation_history:
            messagebox.showwarning("Empty Conversation", "No conversation to save")
            return
        
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = os.path.join(self.conversation_folder, f"conversation_{timestamp}.txt")
        
        try:
            with open(filename, 'w', encoding='utf-8') as f:
                f.write("MEMORY ARK CONVERSATION LOG\n")
                f.write("=" * 50 + "\n\n")
                
                for entry in self.conversation_history:
                    f.write(f"[{entry['timestamp']}] {entry['speaker']}:\n")
                    f.write(f"{entry['message']}\n\n")
            
            messagebox.showinfo("Saved", f"Conversation saved to {filename}")
            self.status_label.config(text=f"Conversation saved")
        except Exception as e:
            messagebox.showerror("Error", f"Couldn't save conversation: {str(e)}")
    
    def clear_conversation(self):
        if messagebox.askyesno("Clear Conversation", "Are you sure? This will clear the AI conversation (document stays)."):
            self.ai_conversation.config(state=tk.NORMAL)
            self.ai_conversation.delete("1.0", tk.END)
            self.ai_conversation.config(state=tk.DISABLED)
            self.conversation_history = []
            self.status_label.config(text="Conversation cleared")

def main():
    root = tk.Tk()
    app = MemoryArkAssistant(root)
    root.mainloop()

if __name__ == "__main__":
    main()
