Your Personal AI Assistant

import sys

# Ollama uses OpenAI-compatible API
client = OpenAI(
    base_url='http://localhost:11434/v1',

class LocalAI:
    def __init__(self):
        self.memory_file = Path.home() / ".config" / "local_ai" / "memory.json"
        self.memory_file.parent.mkdir(parents=True, exist_ok=True)
        self.conversation_history = []
        self.load_memory()
        
    def load_memory(self):
        """Load conversation history from file"""
        if self.memory_file.exists():
            with open(self.memory_file, 'r') as f:
                data = json.load(f)
                self.conversation_history = data.get('history', [])
                if self.conversation_history:
                    print(f"✓ Loaded {len(self.conversation_history)} previous messages")
    
    def save_memory(self):
        """Save conversation history to file"""
        with open(self.memory_file, 'w') as f:
            json.dump({
                'history': self.conversation_history,
                'last_updated': datetime.now().isoformat()
            }, f, indent=2)
    
    def chat(self, user_message):
        """Chat with AI, maintaining conversation history"""
        
        # Add user message to history
        self.conversation_history.append({
            "role": "user",
            "content": user_message
        })
        
        # Get response from Ollama
        try:
            response = client.chat.completions.create(
                model="llama3.2",
                messages=self.conversation_history
            )
            
            # Extract AI response
            ai_message = response.choices[0].message.content
            
            # Add to history
            self.conversation_history.append({
                "role": "assistant",
                "content": ai_message
            })
            
            # Save to disk
            self.save_memory()
            
            return ai_message
            
        except Exception as e:
            print(f"\n✗ Error: {e}")
            print("\nMake sure Ollama is running:")
            print("  Windows: Check if Ollama is in your system tray")
            print("\nAnd that you've downloaded the model:")
            print("  ollama pull llama3.2\n")
            return None
    
    def reset(self):
        """Clear conversation history"""
        self.conversation_history = []
        self.save_memory()
        print("✓ Memory cleared")

def main():
    ai = LocalAI()
    
    if len(sys.argv) > 1:
        if sys.argv[1] == "reset":
            ai.reset()
            return
        
        # Use arguments as message
        message = " ".join(sys.argv[1:])
        response = ai.chat(message)
        if response:
            print(f"\nAI: {response}\n")
    else:
        # Interactive mode
        print("="*60)
        print("LOCAL AI - Ready")
        print("="*60)
        print("Type your message and press Enter")
        print("Commands: 'exit' to quit, 'reset' to clear memory")

        while True:

                user_input = input("You: ")
