def log(message):

def capture_desktop():
    try:
        with mss.mss() as sct:

def ocr_image(image_path):
    try:
        text = pytesseract.image_to_string(Image.open(image_path))
        log("OCR extracted text.")
        return text
    except Exception as e:
        log(f"OCR failed: {e}")
        return ""

def ask_local_ai(prompt):
    try:
        response = ollama.generate(model='llama3', prompt=prompt)
        return response['response']
    except Exception as e:
        log(f"AI query failed: {e}")
        return "AI error."

def speak(text):
    engine = pyttsx3.init()
    engine.say(text)
    engine.runAndWait()

# Main Loop (Runs forever; Ctrl+C to stop)
while True:
    image_path = capture_desktop()
    if image_path:
        ocr_text = ocr_image(image_path)
        if ocr_text:

            ai_response = ask_local_ai(ai_prompt)
            log(f"AI says: {ai_response}")
            speak(ai_response)  # Speaks the response
    time.sleep(capture_interval)
